Review #6 的结论很清晰：停止设计，开始构建。

但在构建之前，review 指出了 6 个具体的实现问题需要修复：

1. **Rollback non-atomic** — `git checkout .` 会破坏 sweep 之外的工作
2. **VERIFIER_MODE 未被下游验证** — 导出了但 controller 没有使用
3. **CI script 依赖未文档化的 tracker API**
4. **Redis 版本兼容性** — `-u URI` 需要 Redis 6+
5. **No self-test** — zero 本身没有测试
6. **Oracle confidence gap** — 75% 的检查仍依赖 AI 推理

这些是实现 bug，不是设计问题。逐一修复。

# Fix 1：Rollback Non-Atomic

```bash
# REPLACE the rollback section in phases/p5_regression.sop
# Problem: git checkout . destroys pre-existing uncommitted work
# Solution: stash the PRE-SWEEP state ONCE in Phase 0,
#           retry with targeted revert, final fallback applies pre-sweep stash

## Phase 0 addition — add to p0_bootstrap.sop AFTER lock acquired

create_pre_sweep_checkpoint() {
  cd "$PROJECT"
  
  # Only checkpoint if git repo exists
  if ! git rev-parse --git-dir >/dev/null 2>&1; then
    echo "No git repo — rollback unavailable"
    export ZERO_PRESWEEP_STASH="none"
    export ZERO_PRESWEEP_COMMIT=$(date +%s)  # timestamp as fallback ID
    return 0
  fi
  
  # Stash ALL current state (tracked + untracked) before touching anything
  STASH_MSG="zero-presweep-$(date +%Y%m%d%H%M%S)"
  
  if git diff --quiet && git diff --staged --quiet; then
    # Nothing to stash — record current HEAD as restore point
    export ZERO_PRESWEEP_STASH="none"
    export ZERO_PRESWEEP_COMMIT=$(git rev-parse HEAD)
    echo "Pre-sweep checkpoint: clean tree at $(git rev-parse --short HEAD)"
  else
    # Stash everything including untracked
    git stash push -u -m "$STASH_MSG" 2>/dev/null
    export ZERO_PRESWEEP_STASH=$(git stash list | grep "$STASH_MSG" \
      | head -1 | cut -d: -f1)
    export ZERO_PRESWEEP_COMMIT=$(git rev-parse HEAD)
    echo "Pre-sweep checkpoint: stashed as $ZERO_PRESWEEP_STASH"
  fi
  
  python3 "$ZERO_TRACKER_PY" log "$TRACKER" "{
    \"phase\": 0,
    \"action\": \"presweep_checkpoint\",
    \"stash\": \"$ZERO_PRESWEEP_STASH\",
    \"commit\": \"$ZERO_PRESWEEP_COMMIT\"
  }"
}

create_pre_sweep_checkpoint
```

```bash
# REPLACE the gate fail / rollback section in phases/p5_regression.sop

GATE_ATTEMPT=0
MAX_ATTEMPTS=3

run_regression_gate() {
  GATE_ATTEMPT=$((GATE_ATTEMPT + 1))
  
  cd "$PROJECT"
  npm test 2>&1 | tee "zero_artifacts/gate_attempt_${GATE_ATTEMPT}.txt"
  
  FINAL_FAIL=$(grep -c "FAIL\|failed" \
    "zero_artifacts/gate_attempt_${GATE_ATTEMPT}.txt" 2>/dev/null || echo 0)
  BASELINE_FAIL=$(grep -c "FAIL\|failed" \
    "zero_artifacts/baseline_test.txt" 2>/dev/null || echo 0)
  
  python3 "$ZERO_TRACKER_PY" log "$TRACKER" "{
    \"phase\": 5,
    \"action\": \"gate_attempt\",
    \"attempt\": $GATE_ATTEMPT,
    \"final_fail\": $FINAL_FAIL,
    \"baseline_fail\": $BASELINE_FAIL
  }"
  
  [ "$FINAL_FAIL" -le "$BASELINE_FAIL" ]
}

rollback_to_presweep() {
  cd "$PROJECT"
  
  echo "=== ROLLBACK: restoring pre-sweep state ==="
  
  if [ "$ZERO_PRESWEEP_STASH" = "none" ]; then
    # Clean tree before sweep — reset to pre-sweep commit
    git reset --hard "$ZERO_PRESWEEP_COMMIT" 2>/dev/null \
      && echo "Reset to $ZERO_PRESWEEP_COMMIT" \
      || echo "WARNING: git reset failed — manual intervention needed"
  else
    # There was a stash before sweep — reset HEAD then restore stash
    git reset --hard "$ZERO_PRESWEEP_COMMIT" 2>/dev/null
    git stash apply "$ZERO_PRESWEEP_STASH" 2>/dev/null \
      && git stash drop "$ZERO_PRESWEEP_STASH" 2>/dev/null \
      && echo "Pre-sweep work restored from $ZERO_PRESWEEP_STASH" \
      || echo "WARNING: stash apply failed — $ZERO_PRESWEEP_STASH preserved for manual recovery"
  fi
  
  python3 "$ZERO_TRACKER_PY" log "$TRACKER" "{
    \"phase\": 5,
    \"action\": \"rollback_complete\",
    \"restored_to\": \"$ZERO_PRESWEEP_COMMIT\",
    \"note\": \"fixes from sweep reverted — bug reports preserved in tracker\"
  }"
  
  echo "Rollback complete."
  echo "Bug reports still in: $TRACKER"
  echo "Zero report still in: $PROJECT/zero_report.md"
}

# Gate loop
while ! run_regression_gate; do
  if [ "$GATE_ATTEMPT" -ge "$MAX_ATTEMPTS" ]; then
    echo "Gate failed $MAX_ATTEMPTS times. Rolling back."
    rollback_to_presweep
    
    # Still run P6 — report gaps, but mark no fixes committed
    python3 "$ZERO_TRACKER_PY" log "$TRACKER" '{
      "phase": 5,
      "action": "gate_failed_rollback",
      "fixes_committed": false
    }'
    
    export ZERO_FIXES_COMMITTED=false
    break
  fi
  
  # Not at max yet — identify which tests newly failed
  NEWLY_FAILED=$(diff \
    <(grep "FAIL" zero_artifacts/baseline_test.txt 2>/dev/null | sort) \
    <(grep "FAIL" "zero_artifacts/gate_attempt_${GATE_ATTEMPT}.txt" | sort) \
    | grep "^>" | sed 's/^> //')
  
  echo "Newly failing tests:"
  echo "$NEWLY_FAILED"
  
  python3 "$ZERO_TRACKER_PY" log "$TRACKER" "{
    \"phase\": 5,
    \"action\": \"returning_to_p3\",
    \"attempt\": $GATE_ATTEMPT,
    \"newly_failed\": \"$(echo "$NEWLY_FAILED" | head -5 | tr '\n' '|')\"
  }"
  
  # Controller transitions to P3 for targeted regression fixes
  echo "SIGNAL:RETURN_TO_P3"
  # After P3 returns, loop re-runs gate
done
```

```python
# ADD these subcommands to zero_tracker.py

import sys
import json
import os
from pathlib import Path


def cmd_get_confidence(tracker_path: str) -> None:
    """
    Reads tracker, computes risk-weighted confidence score.
    Prints integer 0-100 to stdout.
    """
    WEIGHTS = {
        "public_endpoint": 3, "handles_credentials": 3,
        "admin_action": 3, "payment_operation": 3,
        "file_operation": 2, "idor_candidate": 2,
        "database_write": 2, "external_call": 2,
        "background_job": 1
    }
    ORACLE_SCORES = {
        "full": 1.0, "high": 0.85,
        "medium": 0.6, "low": 0.3, "none": 0.0
    }
    # Map tracker statuses to oracle quality
    STATUS_TO_ORACLE = {
        "pass_evidenced": "full",
        "verified": "high",
        "fail_confirmed": "none",
        "suspected_issue": "low",
        "blocked_environment": "none",
        "needs_oracle": "none",
        "unresolvable": "none",
    }

    features = []
    with open(tracker_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)
            if rec.get("kind") == "feature":
                features.append(rec)

    if not features:
        print(0)
        return

    total_weight = 0
    weighted_score = 0.0

    for feat in features:
        flags = feat.get("risk_flags", [])
        weight = max((WEIGHTS.get(f, 1) for f in flags), default=1)

        status = feat.get("status", "needs_oracle")
        oracle_quality = STATUS_TO_ORACLE.get(status, "none")
        score = ORACLE_SCORES.get(oracle_quality, 0.0)

        weighted_score += weight * score
        total_weight += weight

    confidence = int((weighted_score / total_weight * 100) if total_weight > 0 else 0)
    print(confidence)


def cmd_count_issues(tracker_path: str, filter_str: str) -> None:
    """
    Counts issues matching filter.
    Filter format: "status=fail_confirmed severity=high"
    Prints integer count to stdout.
    """
    filters = {}
    for part in filter_str.split():
        if "=" in part:
            k, v = part.split("=", 1)
            filters[k] = v.split(",")  # support comma-separated values

    count = 0
    with open(tracker_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)
            if rec.get("kind") != "issue":
                continue

            match = True
            for k, allowed_values in filters.items():
                rec_val = str(rec.get(k, ""))
                if rec_val not in allowed_values:
                    match = False
                    break

            if match:
                count += 1

    print(count)


def cmd_get_meta(tracker_path: str, key: str) -> None:
    """
    Reads a metadata value from the tracker's meta record.
    Prints value to stdout, empty string if not found.
    """
    with open(tracker_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)
            if rec.get("kind") == "meta":
                print(rec.get(key, ""))
                return
    print("")


def cmd_list_issues(tracker_path: str, filter_str: str) -> None:
    """
    Lists issues matching filter.
    Prints one line per matching issue: id | feature_id | severity | title
    """
    filters = {}
    for part in filter_str.split():
        if "=" in part:
            k, v = part.split("=", 1)
            filters[k] = v.split(",")

    with open(tracker_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)
            if rec.get("kind") != "issue":
                continue

            match = True
            for k, allowed_values in filters.items():
                rec_val = str(rec.get(k, ""))
                if rec_val not in allowed_values:
                    match = False
                    break

            if match:
                print(f"{rec.get('issue_id','?')} | "
                      f"{rec.get('feature_id','?')} | "
                      f"{rec.get('severity','?')} | "
                      f"{rec.get('title','?')}")


def cmd_export_verified(tracker_path: str) -> None:
    """
    Exports all verified features with their reproduce_cmd.
    Outputs JSON array to stdout.
    """
    verified = []
    with open(tracker_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)
            if rec.get("kind") == "feature" and rec.get("status") == "verified":
                verified.append({
                    "feature_id": rec["id"],
                    "name": rec.get("name", ""),
                    "reproduce_cmd": rec.get("reproduce_cmd", ""),
                    "evidence_artifact": rec.get("evidence_artifact", "")
                })
    print(json.dumps(verified, indent=2))


def cmd_summary(tracker_path: str) -> None:
    """
    Prints human-readable summary for CI logs.
    """
    features = {"total": 0, "pass": 0, "fail": 0, "verified": 0, "blocked": 0, "other": 0}
    issues = {"critical": 0, "high": 0, "medium": 0, "low": 0}

    with open(tracker_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)

            if rec.get("kind") == "feature":
                features["total"] += 1
                status = rec.get("status", "")
                if status in ("pass_evidenced",):
                    features["pass"] += 1
                elif status == "verified":
                    features["verified"] += 1
                elif status in ("fail_confirmed", "regressed"):
                    features["fail"] += 1
                elif status == "blocked_environment":
                    features["blocked"] += 1
                else:
                    features["other"] += 1

            elif rec.get("kind") == "issue":
                if rec.get("status") == "fail_confirmed":
                    sev = rec.get("severity", "low")
                    if sev in issues:
                        issues[sev] += 1

    print(f"Features: {features['total']} total | "
          f"{features['pass']} pass | "
          f"{features['verified']} verified | "
          f"{features['fail']} fail | "
          f"{features['blocked']} blocked")
    print(f"Open issues: {issues['critical']} critical | "
          f"{issues['high']} high | "
          f"{issues['medium']} medium | "
          f"{issues['low']} low")


# Add to main() dispatch in zero_tracker.py:
COMMANDS = {
    "get-confidence": (cmd_get_confidence, ["tracker_path"]),
    "count-issues":   (cmd_count_issues,   ["tracker_path", "filter_str"]),
    "get-meta":       (cmd_get_meta,       ["tracker_path", "key"]),
    "list-issues":    (cmd_list_issues,    ["tracker_path", "filter_str"]),
    "export-verified":(cmd_export_verified,["tracker_path"]),
    "summary":        (cmd_summary,        ["tracker_path"]),
}
```

Updated CI script (removes `find /`):

```bash
#!/usr/bin/env bash
# zero_ci_check.sh v2
# Usage: bash zero_ci_check.sh <project_path> [options]
#
# Options:
#   --min-confidence=N    Minimum confidence % (default: 70)
#   --fail-on=SEVERITY    Fail if unresolved issue >= severity (default: high)
#   --strict              Equivalent to --min-confidence=85 --fail-on=low
#   --tracker=PATH        Override tracker path
#   --help                Show this message

set -euo pipefail

show_help() {
  grep "^#" "$0" | grep -v "^#!/" | sed 's/^# //'
  exit 0
}

# Defaults
PROJECT=""
MIN_CONFIDENCE=70
FAIL_ON_SEVERITY="high"
TRACKER_OVERRIDE=""
FAIL_ON_UNRESOLVED="${ZERO_FAIL_ON_UNRESOLVED:-false}"
REQUIRE_BASELINE="${ZERO_REQUIRE_BASELINE:-true}"

# Parse args
for arg in "$@"; do
  case "$arg" in
    --help|-h)             show_help ;;
    --min-confidence=*)    MIN_CONFIDENCE="${arg#*=}" ;;
    --fail-on=*)           FAIL_ON_SEVERITY="${arg#*=}" ;;
    --strict)              MIN_CONFIDENCE=85; FAIL_ON_SEVERITY="low" ;;
    --tracker=*)           TRACKER_OVERRIDE="${arg#*=}" ;;
    -*)                    echo "Unknown option: $arg"; show_help ;;
    *)                     PROJECT="$arg" ;;
  esac
done

if [ -z "$PROJECT" ]; then
  echo "ERROR: project_path required"
  show_help
fi

if [ ! -d "$PROJECT" ]; then
  echo "ERROR: Project directory not found: $PROJECT"
  exit 2
fi

# Validate mutually exclusive severity levels
VALID_SEVERITIES="critical high medium low"
if ! echo "$VALID_SEVERITIES" | grep -qw "$FAIL_ON_SEVERITY"; then
  echo "ERROR: --fail-on must be one of: $VALID_SEVERITIES"
  exit 2
fi

# Locate tracker
if [ -n "$TRACKER_OVERRIDE" ]; then
  TRACKER="$TRACKER_OVERRIDE"
else
  TRACKER="$PROJECT/zero_tracker.jsonl"
fi

if [ ! -f "$TRACKER" ]; then
  echo "ERROR: Tracker not found: $TRACKER"
  echo "Run: /zero --project $PROJECT"
  exit 2
fi

# Locate zero_tracker.py — use known paths, not find /
ZERO_TRACKER_PY=""
for candidate in \
  "${ZERO_TRACKER_PY_PATH:-}" \
  "$(dirname "$0")/../memory/zero_tracker.py" \
  "$(dirname "$0")/../../memory/zero_tracker.py" \
  "${HOME}/.config/ga/memory/zero_tracker.py" \
  "/workspace/memory/zero_tracker.py"
do
  if [ -n "$candidate" ] && [ -f "$candidate" ]; then
    ZERO_TRACKER_PY="$candidate"
    break
  fi
done

if [ -z "$ZERO_TRACKER_PY" ]; then
  echo "ERROR: zero_tracker.py not found."
  echo "Set ZERO_TRACKER_PY_PATH=/path/to/zero_tracker.py"
  exit 2
fi

# Helper: run tracker command
tracker() {
  python3 "$ZERO_TRACKER_PY" "$@" 2>/dev/null || echo ""
}

# Severity rank (lower = more severe)
severity_rank() {
  case "$1" in
    critical) echo 1 ;;
    high)     echo 2 ;;
    medium)   echo 3 ;;
    low)      echo 4 ;;
    *)        echo 99 ;;
  esac
}

FAIL_RANK=$(severity_rank "$FAIL_ON_SEVERITY")
GATE_FAILED=false
EXIT_CODE=0

echo "=== Zero CI Quality Gate ==="
echo "Project:      $PROJECT"
echo "Tracker:      $TRACKER"
echo "Min confidence: ${MIN_CONFIDENCE}%"
echo "Fail on:      >= ${FAIL_ON_SEVERITY}"
echo ""

# Gate 1: Confidence
CONFIDENCE=$(tracker get-confidence "$TRACKER")
CONFIDENCE=${CONFIDENCE:-0}
echo "Confidence: ${CONFIDENCE}% (min: ${MIN_CONFIDENCE}%)"
if [ "$CONFIDENCE" -lt "$MIN_CONFIDENCE" ]; then
  echo "  → FAIL"
  GATE_FAILED=true
  EXIT_CODE=1
else
  echo "  → PASS"
fi

# Gate 2: Unresolved severity
echo ""
echo "Unresolved issues:"
for severity in critical high medium low; do
  COUNT=$(tracker count-issues "$TRACKER" "status=fail_confirmed severity=$severity")
  COUNT=${COUNT:-0}
  SRANK=$(severity_rank "$severity")
  
  MARKER=""
  if [ "$COUNT" -gt 0 ] && [ "$SRANK" -le "$FAIL_RANK" ]; then
    MARKER=" → FAIL"
    GATE_FAILED=true
    EXIT_CODE=1
  elif [ "$COUNT" -gt 0 ]; then
    MARKER=" → WARN"
  fi
  
  echo "  $severity: $COUNT${MARKER}"
done

# Gate 3: Baseline
if [ "$REQUIRE_BASELINE" = "true" ]; then
  echo ""
  BASELINE=$(tracker get-meta "$TRACKER" "baseline_tests_passing")
  FINAL=$(tracker get-meta "$TRACKER" "final_tests_passing")
  BASELINE=${BASELINE:-0}
  FINAL=${FINAL:-0}
  
  echo "Test baseline: $BASELINE → $FINAL"
  if [ "$FINAL" -lt "$BASELINE" ]; then
    echo "  → FAIL (regression)"
    GATE_FAILED=true
    EXIT_CODE=1
  else
    echo "  → PASS"
  fi
fi

# Gate 4: Sweep status
echo ""
STATUS=$(tracker get-meta "$TRACKER" "sweep_status")
STATUS=${STATUS:-unknown}
echo "Sweep status: $STATUS"
case "$STATUS" in
  complete)   echo "  → PASS" ;;
  suspended)  echo "  → FAIL (incomplete)"; GATE_FAILED=true; EXIT_CODE=1 ;;
  aborted)    echo "  → FAIL (aborted)"; EXIT_CODE=2 ;;
  *)          echo "  → WARN (unknown)" ;;
esac

# Gate 5: High-risk coverage gaps
echo ""
UNCOVERED=$(tracker count-issues "$TRACKER" \
  "status=needs_oracle,blocked_environment risk_flag=payment_operation,admin_action,handles_credentials")
UNCOVERED=${UNCOVERED:-0}
echo "High-risk features uncovered: $UNCOVERED"
if [ "$UNCOVERED" -gt 0 ]; then
  echo "  → WARN"
  tracker list-issues "$TRACKER" \
    "status=needs_oracle,blocked_environment \
     risk_flag=payment_operation,admin_action,handles_credentials" \
    | sed 's/^/    /'
  if [ "$FAIL_ON_SEVERITY" = "low" ]; then
    echo "  → FAIL (strict mode)"
    GATE_FAILED=true
    EXIT_CODE=1
  fi
fi

# Summary
echo ""
echo "=== Summary ==="
```

# Fix 4：Redis 版本兼容性

```bash
# REPLACE redis section in lock implementation

connect_redis() {
  local URL="${ZERO_REDIS_URL:-redis://localhost:6379}"
  
  # Detect Redis version
  # Try URI format first (Redis 6+), fall back to -h -p (Redis 4+)
  REDIS_VERSION=$(redis-cli -u "$URL" INFO server 2>/dev/null \
    | grep "redis_version" | cut -d: -f2 | tr -d '[:space:]')
  
  if [ -n "$REDIS_VERSION" ]; then
    # URI format works — Redis 6+
    REDIS_CMD="redis-cli -u $URL"
    echo "Redis $REDIS_VERSION connected (URI mode)"
    return 0
  fi
  
  # URI failed — parse and try -h -p [-a] format
  # Parse redis://[:password@]host[:port][/db]
  REDIS_HOST=$(echo "$URL" | sed 's|redis://||' | sed 's|.*@||' | cut -d: -f1 | cut -d/ -f1)
  REDIS_PORT=$(echo "$URL" | sed 's|redis://||' | sed 's|.*@||' | cut -d: -f2 | cut -d/ -f1)
  REDIS_PASS=$(echo "$URL" | grep "@" | sed 's|redis://||' | sed 's|@.*||' | cut -d: -f2)
  
  REDIS_HOST="${REDIS_HOST:-localhost}"
  REDIS_PORT="${REDIS_PORT:-6379}"
  
  if [ -n "$REDIS_PASS" ]; then
    REDIS_CMD="redis-cli -h $REDIS_HOST -p $REDIS_PORT -a $REDIS_PASS"
  else
    REDIS_CMD="redis-cli -h $REDIS_HOST -p $REDIS_PORT"
  fi
  
  # Test connection
  REDIS_VERSION=$($REDIS_CMD INFO server 2>/dev/null \
    | grep "redis_version" | cut -d: -f2 | tr -d '[:space:]')
  
  if [ -n "$REDIS_VERSION" ]; then
    echo "Redis $REDIS_VERSION connected (host:port mode)"
    return 0
  fi
  
  echo "ERROR: Cannot connect to Redis at $URL"
  return 1
}

acquire_lock_redis() {
  if ! connect_redis; then
    echo "WARNING: Redis unavailable. Falling back to file lock."
    LOCK_METHOD="file"
    acquire_lock_file
    return $?
  fi
  
  # SET NX PX (milliseconds) — supported in Redis 2.6.12+
  # Preferred over SET NX EX for finer granularity
  LOCK_TTL_MS=$(( 4 * 60 * 60 * 1000 ))  # 4 hours in ms
  LOCK_VALUE="$$@$(hostname)@$(date +%s)"
  
  RESULT=$($REDIS_CMD SET "$LOCK_KEY" "$LOCK_VALUE" NX PX "$LOCK_TTL_MS" 2>/dev/null)
  
  if [ "$RESULT" = "OK" ]; then
    echo "Lock acquired (redis, key: $LOCK_KEY)"
    trap "$REDIS_CMD DEL $LOCK_KEY >/dev/null 2>&1; release_log" EXIT INT TERM
    return 0
  else
    HOLDER=$($REDIS_CMD GET "$LOCK_KEY" 2>/dev/null || echo "unknown")
    echo "ERROR: Lock held by: $HOLDER"
    return 1
  fi
}
```

```bash
check_results() {
  TRACKER="$FIXTURE_DIR/zero_tracker.jsonl"
  
  if [ ! -f "$TRACKER" ]; then
    echo "FAIL: Tracker not created — zero did not run successfully"
    return 1
  fi

import json, sys
keyword = sys.argv[1].lower()
with open('$TRACKER') as f:
    for line in f:
        rec = json.loads(line)
        if rec.get('kind') == 'issue':
            text = json.dumps(rec).lower()
            if keyword in text:
                sys.exit(0)
sys.exit(1)
" "$FINDING_ID" 2>/dev/null; then
      FOUND=$((FOUND + 1))

  echo "=== Smoke Test Results ==="
  echo "Expected: $EXPECTED_COUNT findings"
  echo "Found:    $FOUND findings"
  echo "Missed:   ${#MISSED[@]} findings"

  FP_COUNT=$(python3 -c "
import json
count = 0
expected_ids = set('$( echo "${!EXPECTED_FINDINGS[@]}" )'.split())
with open('$TRACKER') as f:
    for line in f:
        rec = json.loads(line)
        if rec.get('kind') == 'issue' and rec.get('status') == 'fail_confirmed':
            found = False
            text = json.dumps(rec).lower()
            for eid in expected_ids:
                if eid.lower() in text:
                    found = True
                    break
            if not found:
                count += 1
print(count)
" 2>/dev/null || echo 0)
  
  echo "False positives (unexpected confirmed findings): $FP_COUNT"

  # - Found at least 80% of expected findings (some may be env-blocked)
  # - False positives <= 2

  else
    echo ""
    echo "RESULT: FAIL"
    if [ "$FOUND" -lt "$MIN_FOUND" ]; then
      echo "  Reason: Only found $FOUND/$EXPECTED_COUNT (min: $MIN_FOUND)"
      for m in "${MISSED[@]}"; do
        echo "    - Missed: $m"
      done
    fi
    if [ "$FP_COUNT" -gt 2 ]; then
      echo "  Reason: $FP_COUNT false positives (max: 2)"
    fi
    return 1
  fi
}
```

# Fix 6：Oracle Confidence Gap（将 suspected 转为 test）

这个是最根本的问题：75% 的 inpval 检查依赖 AI 推理。解决方案是给每个 suspected check 强制生成一个测试，让通过/失败决定结论。

在 `phases/p3_batch_loop.sop` 加入：

```bash
## 3.3b Suspected Check → Mandatory Test Generation

# For every suspected_issue from triage that has NOT been trace-confirmed:
# MUST generate a test before it can remain as suspected or be cleared.
# "Read code and think it's safe" is NOT sufficient to clear a suspicion.

promote_suspected_to_tested() {
  local ISSUE_ID="$1"
  local FEATURE_ID="$2"
  local CHECK_ID="$3"    # e.g., INPVAL-01
  local ORACLE_ID="$4"  # e.g., security/inpval

  # Step 1: Generate targeted test for this specific check
  # Load the test template for this check from oracle file
  TEMPLATE=$(python3 -c "
import re, sys
oracle = open('skills/zero/oracles/$ORACLE_ID.oracle').read()
# Find the test template section matching CHECK_ID
pattern = rf'// {re.escape(\"$CHECK_ID\")}.*?(?=// [A-Z]|\Z)'
match = re.search(pattern, oracle, re.DOTALL)
if match:
    print(match.group(0)[:2000])  # cap at 2000 chars
else:
    print('')
" 2>/dev/null)

  if [ -z "$TEMPLATE" ]; then
    # No template for this specific check
    # Generate minimal smoke test based on check category
    generate_minimal_test "$CHECK_ID" "$FEATURE_ID"
  else
    # Adapt template to this project's conventions
    adapt_test_template "$TEMPLATE" "$FEATURE_ID" "$CHECK_ID"
  fi

  # Step 2: Run the generated test
  TEST_FILE="zero_artifacts/batches/${CURRENT_BATCH}/tests/${CHECK_ID}_${FEATURE_ID}.test.ts"
  
  if [ -f "$TEST_FILE" ]; then
    npx jest "$TEST_FILE" 2>&1 \
      | tee "zero_artifacts/batches/${CURRENT_BATCH}/tests/${CHECK_ID}_${FEATURE_ID}.result.txt"
    TEST_EXIT=${PIPESTATUS[0]}
    
    if [ "$TEST_EXIT" -eq 0 ]; then
      # Test passed → check is clear
      python3 "$ZERO_TRACKER_PY" update "$TRACKER" "$ISSUE_ID" \
        status "false_positive" \
        evidence_type "generated_test_pass" \
        evidence_artifact "$TEST_FILE.result.txt"
    else
      # Test failed → suspected becomes confirmed
      python3 "$ZERO_TRACKER_PY" update "$TRACKER" "$ISSUE_ID" \
        status "fail_confirmed" \
        confidence "confirmed" \
        evidence_type "generated_test_fail" \
        evidence_artifact "$TEST_FILE.result.txt"
    fi
  else
    # Could not generate test — remains suspected with note
    python3 "$ZERO_TRACKER_PY" update "$TRACKER" "$ISSUE_ID" \
      oracle_note "test_generation_failed — oracle template not found for $CHECK_ID"
  fi
}

# Rule: no suspected_issue may exit Phase 3 without one of:
# a) trace-confirmed (evidence_type = trace_confirmed)
# b) test-confirmed (evidence_type = generated_test_fail)
# c) test-cleared   (evidence_type = generated_test_pass)
# d) explicitly marked: test_generation_failed + reason

# Check at batch end:
UNRESOLVED_SUSPECTED=$(python3 "$ZERO_TRACKER_PY" count-issues "$TRACKER" \
  "status=suspected_issue batch=$CURRENT_BATCH" 2>/dev/null || echo 0)

if [ "$UNRESOLVED_SUSPECTED" -gt 0 ]; then
  echo "WARNING: $UNRESOLVED_SUSPECTED suspected issues remain without test or trace."
  echo "These reduce oracle confidence score."
  python3 "$ZERO_TRACKER_PY" list-issues "$TRACKER" \
    "status=suspected_issue batch=$CURRENT_BATCH"
fi
```
