## Self-Validation (mandatory before proceeding)

```bash
# Confirm user input entry points exist
grep -rn $INCLUDE $EXCLUDE \
  -e "req\.body\|req\.params\|req\.query\|req\.headers\|request\.form\|request\.args\|request\.json\|ctx\.request" \
  $PROJECT 2>/dev/null | head -10
```

Before proceeding, identify:
- How many routes accept user input (body/params/query)?
- Is there a shared validation middleware (zod/joi/yup/pydantic/marshmallow)?
- Is validation applied at route level or controller level?

If zero input-accepting routes found → not applicable, skip.
If shared validator exists → checks focus on whether it is applied consistently.
If no shared validator → checks focus on per-route validation gaps.

# 补全 2：Rollback (git stash before Phase 5 retry)

这段加入 `phases/p5_regression.sop` 的 Phase 5 Gate Fail 分支：

```bash
# Add to p5_regression.sop — replace gate fail section

## 5.3 Gate Rules with Rollback

GATE_ATTEMPT=1

run_regression_gate() {
  # Re-run full test suite
  cd $PROJECT
  npm test 2>&1 | tee zero_artifacts/final_test_attempt_${GATE_ATTEMPT}.txt
  FINAL_FAIL=$(grep -c "FAIL\|failed" \
    zero_artifacts/final_test_attempt_${GATE_ATTEMPT}.txt || echo 0)
  BASELINE_FAIL=$(grep -c "FAIL\|failed" zero_artifacts/baseline_test.txt || echo 0)
  
  if [ "$FINAL_FAIL" -le "$BASELINE_FAIL" ]; then
    echo "GATE_PASS"
    return 0
  else
    echo "GATE_FAIL: $FINAL_FAIL failures vs baseline $BASELINE_FAIL"
    return 1
  fi
}

# Stash current state before first retry attempt
stash_before_retry() {
  cd $PROJECT
  git stash push -m "zero-sweep-attempt-${GATE_ATTEMPT}-$(date +%Y%m%d%H%M%S)" \
    2>/dev/null && echo "STASHED" || echo "STASH_FAILED_NO_GIT"
}

pop_stash() {
  cd $PROJECT
  git stash pop 2>/dev/null && echo "STASH_POPPED" || echo "STASH_POP_FAILED"
}

# Gate execution with retry loop
if run_regression_gate; then
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"gate_pass","attempt":1}'
else
  # Attempt 1 failed
  GATE_ATTEMPT=2
  
  # Stash current (possibly broken) state
  STASH_NAME=$(stash_before_retry)
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"gate_fail","attempt":1,"stash":"'"$STASH_NAME"'"}'
  
  # Return to batch loop for regressions
  # (controller transitions to P3_BATCH_LOOP via state machine)
  echo "RETURN_TO_PHASE_3"
  exit 0  # controller handles transition
fi

# After Phase 3 regression fixes, re-run gate
if run_regression_gate; then
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"gate_pass","attempt":2}'
else
  # Attempt 2 failed
  GATE_ATTEMPT=3
  STASH_NAME=$(stash_before_retry)
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"gate_fail","attempt":2,"stash":"'"$STASH_NAME"'"}'
  echo "RETURN_TO_PHASE_3"
  exit 0
fi

# After second Phase 3 pass, run gate one final time
if run_regression_gate; then
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"gate_pass","attempt":3}'
else
  # Attempt 3 failed — rollback to pre-sweep state
  GATE_ATTEMPT="FINAL_FAIL"
  
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"gate_fail_rollback","attempt":3}'
  
  # Hard rollback: pop all zero-sweep stashes
  echo "All 3 gate attempts failed. Rolling back sweep changes."
  
  cd $PROJECT
  git stash list | grep "zero-sweep-attempt" | while read -r entry; do
    STASH_REF=$(echo "$entry" | cut -d: -f1)
    git stash drop "$STASH_REF" 2>/dev/null
  done
  
  # Restore to pre-sweep HEAD
  git checkout . 2>/dev/null || echo "ROLLBACK_FAILED: manual intervention needed"
  
  bash $ZERO_SH --log $PROJECT \
    '{"phase":5,"action":"rollback_complete","note":"sweep reverted, proceed to P6 for gap report only"}'
  
  # Still run P6 — confidence report reflects 0 fixes committed
  echo "PROCEED_TO_P6_WITH_ROLLBACK"
fi
```

## 0.1b Subagent Capability Probe

# Determine what execution modes are available
# This affects: verifier (Phase 6), hive workers (Phase 3), goal mode

SUBAGENT_CAPABLE=false
HIVE_CAPABLE=false
GOAL_CAPABLE=false

# Probe 1: Can we launch a subagent via boss.sh?

    SUBAGENT_CAPABLE=true
  fi
  
  if echo "$BOSS_PROBE" | grep -q "hive.*true\|can_hive.*true"; then
    HIVE_CAPABLE=true
  fi
  
  if echo "$BOSS_SH" | grep -q "goal.*true\|can_goal.*true"; then
    GOAL_CAPABLE=true
  fi
fi

# Probe 2: Minimal subagent smoke test (if probe 1 unclear)
if [ "$SUBAGENT_CAPABLE" = "false" ] && [ -n "$BOSS_SH" ]; then
  SMOKE=$(bash $BOSS_SH "echo zero-capability-test" 2>/dev/null)
  if [ -n "$SMOKE" ]; then
    SUBAGENT_CAPABLE=true
  fi

# Record capabilities
python3 $ZERO_TRACKER_PY log $TRACKER '{
  "phase": 0,
  "action": "capability_probe",
  "subagent_capable": '"$SUBAGENT_CAPABLE"',
  "hive_capable": '"$HIVE_CAPABLE"',
  "goal_capable": '"$GOAL_CAPABLE"'

# Pre-set verifier mode for Phase 6

  VERIFIER_MODE="independent_agent"

  VERIFIER_MODE="self_verify_degraded"

# 补全 4：zero_ci_check.sh

```bash
#!/usr/bin/env bash
# zero_ci_check.sh
# Usage: bash zero_ci_check.sh <project_path> [--min-confidence=70] [--fail-on=medium]
# Returns: 0 (pass) | 1 (fail) | 2 (config error)
#
# Designed for: GitHub Actions, GitLab CI, Jenkins, any CI that checks exit code
#
# Example GitHub Actions step:
#   - name: Zero Quality Gate
#     run: bash skills/zero/zero_ci_check.sh ${{ github.workspace }} --min-confidence=75
#     env:
#       ZERO_FAIL_ON_UNRESOLVED: "true"

set -euo pipefail

PROJECT="${1:-}"
MIN_CONFIDENCE=70
FAIL_ON_SEVERITY="high"      # fail if any unresolved issue >= this severity
FAIL_ON_UNRESOLVED="${ZERO_FAIL_ON_UNRESOLVED:-false}"
REQUIRE_BASELINE_MAINTAINED="${ZERO_REQUIRE_BASELINE:-true}"

# Parse args
for arg in "${@:2}"; do
  case "$arg" in
    --min-confidence=*) MIN_CONFIDENCE="${arg#*=}" ;;
    --fail-on=*)        FAIL_ON_SEVERITY="${arg#*=}" ;;
    --strict)           FAIL_ON_SEVERITY="low"; MIN_CONFIDENCE=85 ;;
  esac
done

# Validation
if [ -z "$PROJECT" ] || [ ! -d "$PROJECT" ]; then
  echo "ERROR: Project path required and must exist"
  exit 2
fi

TRACKER="$PROJECT/zero_tracker.jsonl"
REPORT="$PROJECT/zero_report.md"

if [ ! -f "$TRACKER" ]; then
  echo "ERROR: No zero sweep found at $PROJECT"
  echo "Run: /zero --project $PROJECT"
  exit 2
fi

# Find zero_tracker.py
ZERO_TRACKER_PY=$(find / -name "zero_tracker.py" -path "*/memory/*" 2>/dev/null | head -1)
if [ -z "$ZERO_TRACKER_PY" ]; then
  echo "ERROR: zero_tracker.py not found"
  exit 2
fi

echo "=== Zero CI Quality Gate ==="
echo "Project: $PROJECT"
echo "Tracker: $TRACKER"
echo ""

GATE_FAILED=false
EXIT_CODE=0

# Check 1: Confidence Score
CONFIDENCE=$(python3 "$ZERO_TRACKER_PY" get-confidence "$TRACKER" 2>/dev/null || echo "0")
echo "Confidence Score: ${CONFIDENCE}%"
echo "Minimum Required: ${MIN_CONFIDENCE}%"

if [ "$CONFIDENCE" -lt "$MIN_CONFIDENCE" ]; then
  echo "FAIL: Confidence ${CONFIDENCE}% below minimum ${MIN_CONFIDENCE}%"
  GATE_FAILED=true
  EXIT_CODE=1
else
  echo "PASS: Confidence threshold met"
fi

echo ""

# Check 2: Unresolved Issues by Severity
for severity in critical high medium low; do
  COUNT=$(python3 "$ZERO_TRACKER_PY" count-issues "$TRACKER" \
    "status=fail_confirmed severity=$severity" 2>/dev/null || echo "0")
  
  echo "Unresolved $severity issues: $COUNT"
  
  # Check if this severity triggers failure
  SEVERITY_ORDER="critical high medium low"
  FAIL_SEVERITY_RANK=$(echo "$SEVERITY_ORDER" | tr ' ' '\n' | \
    grep -n "^${FAIL_ON_SEVERITY}$" | cut -d: -f1)
  CURRENT_RANK=$(echo "$SEVERITY_ORDER" | tr ' ' '\n' | \
    grep -n "^${severity}$" | cut -d: -f1)
  
  if [ "$COUNT" -gt "0" ] && [ "$CURRENT_RANK" -le "$FAIL_SEVERITY_RANK" ]; then
    echo "FAIL: $COUNT unresolved $severity issue(s)"
    GATE_FAILED=true
    EXIT_CODE=1
  fi
done

echo ""

# Check 3: Baseline Maintained
if [ "$REQUIRE_BASELINE_MAINTAINED" = "true" ]; then
  BASELINE_TESTS=$(python3 "$ZERO_TRACKER_PY" get-meta "$TRACKER" \
    "baseline_tests_passing" 2>/dev/null || echo "0")
  FINAL_TESTS=$(python3 "$ZERO_TRACKER_PY" get-meta "$TRACKER" \
    "final_tests_passing" 2>/dev/null || echo "0")
  
  echo "Tests passing: $BASELINE_TESTS → $FINAL_TESTS"
  
  if [ "$FINAL_TESTS" -lt "$BASELINE_TESTS" ]; then
    echo "FAIL: Test regression — fewer tests passing than baseline"
    GATE_FAILED=true
    EXIT_CODE=1
  else
    echo "PASS: Baseline maintained or improved"
  fi
fi

echo ""

# Check 4: High-Risk Features Covered
UNCOVERED_HIGH_RISK=$(python3 "$ZERO_TRACKER_PY" count-issues "$TRACKER" \
  "coverage=none risk_flag=payment_operation,admin_action,handles_credentials" \
  2>/dev/null || echo "0")

if [ "$UNCOVERED_HIGH_RISK" -gt "0" ]; then
  echo "WARNING: $UNCOVERED_HIGH_RISK high-risk feature(s) with no coverage"
  python3 "$ZERO_TRACKER_PY" list-issues "$TRACKER" \
    "coverage=none risk_flag=payment_operation,admin_action,handles_credentials" \
    2>/dev/null
  # Warning, not failure (unless strict mode)
  if [ "$FAIL_ON_SEVERITY" = "low" ]; then
    GATE_FAILED=true
    EXIT_CODE=1
  fi
fi

echo ""

# Check 5: Sweep Completed (not suspended/aborted)
SWEEP_STATUS=$(python3 "$ZERO_TRACKER_PY" get-meta "$TRACKER" \
  "sweep_status" 2>/dev/null || echo "unknown")

case "$SWEEP_STATUS" in
  "complete")
    echo "PASS: Sweep completed fully"
    ;;
  "suspended")
    echo "FAIL: Sweep was suspended — incomplete results"
    GATE_FAILED=true
    EXIT_CODE=1
    ;;
  "aborted")
    echo "FAIL: Sweep was aborted — no valid results"
    EXIT_CODE=2
    ;;
  *)
    echo "WARNING: Sweep status unknown ($SWEEP_STATUS)"
    ;;
esac

echo ""
echo "=== Summary ==="

if [ "$GATE_FAILED" = "true" ]; then
  echo "RESULT: FAIL"
  echo ""
  echo "Report: $REPORT"
  echo ""
  # Print actionable summary for CI log
  python3 "$ZERO_TRACKER_PY" summary "$TRACKER" 2>/dev/null || true
else
  echo "RESULT: PASS"
  echo "Confidence: ${CONFIDENCE}% (min: ${MIN_CONFIDENCE}%)"
  echo "Report: $REPORT"
fi

exit $EXIT_CODE
```

# 补全 5：ZERO_LOCK_METHOD (多环境 Lock)

加入 `phases/p0_bootstrap.sop` 的 0.1 之前，替换原有 lock 实现：

```bash
## 0.0 Sweep Lock (concurrent sweep safety)

LOCK_METHOD="${ZERO_LOCK_METHOD:-file}"  # file | env | redis
LOCK_KEY="zero_sweep_${PROJECT//\//_}"

acquire_lock() {
  case "$LOCK_METHOD" in
    "file")
      LOCK_FILE="$PROJECT/.zero_sweep.lock"
      
      if [ -f "$LOCK_FILE" ]; then
        LOCK_PID=$(cat "$LOCK_FILE" 2>/dev/null || echo "unknown")
        LOCK_AGE=$(( $(date +%s) - $(stat -c %Y "$LOCK_FILE" 2>/dev/null || echo 0) ))
        
        # Check if PID is still alive (local only — not NFS-safe)
        if [ "$LOCK_PID" != "unknown" ] && kill -0 "$LOCK_PID" 2>/dev/null; then
          echo "ERROR: Active sweep running (PID $LOCK_PID). Aborting."
          return 1
        fi
        
        # Stale lock: file exists but PID dead, or lock > 4 hours old
        if [ "$LOCK_AGE" -gt 14400 ]; then
          echo "WARNING: Stale lock (${LOCK_AGE}s old). Removing."
          rm -f "$LOCK_FILE"
        else
          echo "ERROR: Lock exists (PID $LOCK_PID, ${LOCK_AGE}s old). Use --force to override."
          return 1
        fi
      fi
      
      echo $$ > "$LOCK_FILE"
      trap "rm -f $LOCK_FILE; bash $ZERO_SH --log $PROJECT \
        '{\"action\":\"lock_released\"}'" EXIT INT TERM
      echo "Lock acquired (file, PID $$)"
      ;;
      
    "env")
      # CI-safe: use environment variable (works across NFS, containers)
      # Requires: CI/CD sets ZERO_LOCK_HOLDER externally
      if [ -n "${ZERO_LOCK_HOLDER:-}" ]; then
        echo "ERROR: Lock held by $ZERO_LOCK_HOLDER (env). Aborting."
        return 1
      fi
      export ZERO_LOCK_HOLDER="$$@$(hostname)"
      echo "Lock acquired (env, holder: $ZERO_LOCK_HOLDER)"
      ;;
      
    "redis")
      # NFS-safe, distributed: requires redis-cli
      if ! command -v redis-cli &>/dev/null; then
        echo "WARNING: ZERO_LOCK_METHOD=redis but redis-cli not found. Falling back to file."
        LOCK_METHOD="file"
        acquire_lock
        return $?
      fi
      
      REDIS_URL="${ZERO_REDIS_URL:-redis://localhost:6379}"
      LOCK_TTL=14400  # 4 hours in seconds
      
      # SET NX EX — atomic: set if not exists, with TTL
      RESULT=$(redis-cli -u "$REDIS_URL" \
        SET "$LOCK_KEY" "$$@$(hostname)" NX EX "$LOCK_TTL" 2>/dev/null)
      
      if [ "$RESULT" != "OK" ]; then
        HOLDER=$(redis-cli -u "$REDIS_URL" GET "$LOCK_KEY" 2>/dev/null)
        echo "ERROR: Lock held by $HOLDER (redis). Aborting."
        return 1
      fi
      
      # Release on exit
      trap "redis-cli -u $REDIS_URL DEL $LOCK_KEY >/dev/null 2>&1; \
            bash $ZERO_SH --log $PROJECT '{\"action\":\"lock_released\"}'" \
        EXIT INT TERM
      echo "Lock acquired (redis, key: $LOCK_KEY)"
      ;;
      
    *)
      echo "ERROR: Unknown ZERO_LOCK_METHOD=$LOCK_METHOD. Use: file|env|redis"
      return 1
      ;;
  esac
  
  bash $ZERO_SH --log $PROJECT "{
    \"action\": \"lock_acquired\",
    \"method\": \"$LOCK_METHOD\",
    \"holder\": \"$$@$(hostname)\"
  }"
  return 0
}

release_lock() {
  case "$LOCK_METHOD" in
    "file") rm -f "$PROJECT/.zero_sweep.lock" ;;
    "env")  unset ZERO_LOCK_HOLDER ;;
    "redis")
      REDIS_URL="${ZERO_REDIS_URL:-redis://localhost:6379}"
      redis-cli -u "$REDIS_URL" DEL "$LOCK_KEY" >/dev/null 2>&1
      ;;
  esac
}

# Attempt to acquire lock
if ! acquire_lock; then
  exit 1
fi
```
